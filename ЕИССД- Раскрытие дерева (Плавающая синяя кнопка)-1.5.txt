// ==UserScript==
// @name         ЕИССД: Раскрытие дерева (Кнопка в заголовке)
// @namespace    http://tampermonkey.net
// @version      1.6
// @description  Кнопка «Раскрыть дерево» внутри заголовка «Создание заявки» в ЕИССД.
// @author       Roman Yakovlev
// @match        https://eissd.rt.ru/order/phys/*
// @grant        none
// ==/UserScript==

(function() {
    'use strict';

    let isProcessing = false;

    function expandComboTree() {
        if (isProcessing) return;
        isProcessing = true;

        // Находим зоны плюсиков у неразвернутых веток дерева
        const hitAreas = document.querySelectorAll(
            '.combo-tree .expandable > .hitarea, .treeview .expandable > .hitarea'
        );

        if (hitAreas.length === 0) {
            console.log('[ЕИССД]: Все доступные ветки дерева успешно развернуты.');
            isProcessing = false;
            return;
        }

        hitAreas.forEach(hitArea => {
            const clickEvent = new MouseEvent('click', {
                bubbles: true,
                cancelable: true,
                view: window
            });

            hitArea.dispatchEvent(clickEvent);
        });

        // Задержка на AJAX-подгрузку строк с сервера Ростелекома
        setTimeout(() => {
            isProcessing = false;
            expandComboTree();
        }, 400);
    }

    // Создание кнопки внутри заголовка «Создание заявки»
    function createHeaderButton() {
        // Ищем именно заголовок из указанной разметки
        const header = document.querySelector('h1.c-head-36.js-head');

        if (!header) return;

        // Проверяем дубликаты
        if (document.getElementById('eissd-expand-btn-header')) return;

        const btn = document.createElement('button');
        btn.id = 'eissd-expand-btn-header';
        btn.type = 'button';
        btn.innerText = '🌳 Раскрыть дерево';
        btn.title = 'Раскрыть все доступные ветки дерева';

        // Кнопка находится непосредственно внутри h1
        Object.assign(btn.style, {
            display: 'inline-flex',
            alignItems: 'center',
            verticalAlign: 'middle',
            marginLeft: '18px',
            padding: '7px 14px',
            backgroundColor: '#1a73e8',
            color: '#ffffff',
            border: 'none',
            borderRadius: '4px',
            cursor: 'pointer',
            fontWeight: 'bold',
            fontSize: '12px',
            lineHeight: '1.2',
            fontFamily: 'Arial, sans-serif',
            boxShadow: '0 2px 5px rgba(0,0,0,0.20)',
            transition: 'background-color 0.15s ease, transform 0.1s ease'
        });

        btn.addEventListener('mouseenter', () => {
            btn.style.backgroundColor = '#0b57d0';
        });

        btn.addEventListener('mouseleave', () => {
            btn.style.backgroundColor = '#1a73e8';
            btn.style.transform = 'scale(1)';
        });

        btn.addEventListener('mousedown', () => {
            btn.style.transform = 'scale(0.96)';
        });

        btn.addEventListener('mouseup', () => {
            btn.style.transform = 'scale(1)';
        });

        btn.addEventListener('click', (e) => {
            e.preventDefault();
            e.stopPropagation();

            expandComboTree();
        });

        // Добавляем кнопку в конец h1:
        // <h1 ...>Создание заявки<span class="calcState"></span><button>...</button></h1>
        header.appendChild(btn);

        console.log('[ЕИССД]: Кнопка «Раскрыть дерево» добавлена в заголовок «Создание заявки».');
    }

    // Инициализация
    function init() {
        createHeaderButton();

        // Если заголовок создаётся/перерисовывается сайтом динамически,
        // пробуем добавить кнопку снова после изменения DOM.
        const observer = new MutationObserver(() => {
            createHeaderButton();
        });

        observer.observe(document.body, {
            childList: true,
            subtree: true
        });
    }

    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', init);
    } else {
        init();
    }
})();
